// While Loop

i = 0;
while (i < 5) {
    print i;
    i = i + 1;
}
print "while loop completed";

counter = 0;
while (counter <= 3) {
    print counter;
    counter = counter + 1;
}
print "while loop completed";


// Boolean Expression

a = 5;
b = 10;
if (a == 5) {
    print "a equals 5";
}
if (a != b) {
    print "a is not equal to b";
}
if (a < b) {
    print "a is less than b";
}
if (b > a) {
    print "b is greater than a";
}

// For Loop

for (j = 0; j < 3; j = 1) {
    print j;
    print "inside for loop";
}
print "for loop completed";
